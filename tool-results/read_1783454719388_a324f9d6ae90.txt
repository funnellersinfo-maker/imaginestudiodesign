    60→function FadeUp({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    61→  const ref = useRef(null);
    62→  const inView = useInView(ref, { once: true, margin: "-60px" });
    63→  return (
    64→    <motion.div ref={ref} initial={{ opacity: 0, y: 40 }} animate={inView ? { opacity: 1, y: 0 } : {}}
    65→      transition={{ duration: 0.7, delay, ease: [0.22, 1, 0.36, 1] }} className={className}>{children}</motion.div>
    66→  );
    67→}
    68→function FadeIn({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    69→  const ref = useRef(null);
    70→  const inView = useInView(ref, { once: true, margin: "-40px" });
    71→  return (
    72→    <motion.div ref={ref} initial={{ opacity: 0 }} animate={inView ? { opacity: 1 } : {}}
    73→      transition={{ duration: 0.8, delay, ease: "easeOut" }} className={className}>{children}</motion.div>
    74→  );
    75→}
    76→function ScaleIn({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    77→  const ref = useRef(null);
    78→  const inView = useInView(ref, { once: true, margin: "-40px" });
    79→  return (
    80→    <motion.div ref={ref} initial={{ opacity: 0, scale: 0.9 }} animate={inView ? { opacity: 1, scale: 1 } : {}}
    81→      transition={{ duration: 0.6, delay, ease: [0.22, 1, 0.36, 1] }} className={className}>{children}</motion.div>
    82→  );
    83→}
